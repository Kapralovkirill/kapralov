DROP DATABASE IF EXISTS sports_db;
CREATE DATABASE IF NOT EXISTS sports_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE sports_db;

CREATE TABLE IF NOT EXISTS club (
    id_club INT AUTO_INCREMENT NOT NULL,
    appelation VARCHAR(100) NOT NULL,
    town VARCHAR(50) NOT NULL,
    address VARCHAR(150) NOT NULL,
    CONSTRAINT pk_club PRIMARY KEY (id_club)
);

CREATE TABLE IF NOT EXISTS discharge (
    discharge_code INT AUTO_INCREMENT NOT NULL,
    appelation VARCHAR(50) NOT NULL,
    CONSTRAINT pk_discharge PRIMARY KEY (discharge_code)
);

CREATE TABLE IF NOT EXISTS trainer (
    id_trainer INT AUTO_INCREMENT NOT NULL,
    name VARCHAR(50) NOT NULL,
    surname VARCHAR(50) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    id_club INT NOT NULL,
    CONSTRAINT pk_trainer PRIMARY KEY (id_trainer),
    CONSTRAINT fk_trainer_club FOREIGN KEY (id_club) 
        REFERENCES club(id_club) ON UPDATE CASCADE ON DELETE RESTRICT
);

CREATE TABLE IF NOT EXISTS athlete (
    athlete_code INT AUTO_INCREMENT NOT NULL,
    name VARCHAR(50) NOT NULL,
    surname VARCHAR(50) NOT NULL,
    date_of_birth DATE NOT NULL,
    passport VARCHAR(20) NOT NULL,
    insurance VARCHAR(50) NOT NULL,
    post VARCHAR(50) NULL,
    phone VARCHAR(20) NULL,
    id_club INT NOT NULL,
    id_trainer INT NOT NULL,
    discharge_code INT NOT NULL,
    CONSTRAINT pk_athlete PRIMARY KEY (athlete_code),
    CONSTRAINT uk_athlete_passport UNIQUE (passport),
    CONSTRAINT fk_athlete_club FOREIGN KEY (id_club) 
        REFERENCES club(id_club) ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_athlete_trainer FOREIGN KEY (id_trainer) 
        REFERENCES trainer(id_trainer) ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_athlete_discharge FOREIGN KEY (discharge_code) 
        REFERENCES discharge(discharge_code) ON UPDATE CASCADE ON DELETE RESTRICT
);

CREATE TABLE IF NOT EXISTS competition (
    competition_code INT AUTO_INCREMENT NOT NULL,
    competition_name VARCHAR(100) NOT NULL,
    competition_date DATE NOT NULL,
    town VARCHAR(50) NOT NULL,
    address VARCHAR(150) NOT NULL,
    CONSTRAINT pk_competition PRIMARY KEY (competition_code)
);

CREATE TABLE IF NOT EXISTS result (
    id_result INT AUTO_INCREMENT NOT NULL,
    athlete_code INT NOT NULL,
    competition_code INT NOT NULL,
    category VARCHAR(30) NOT NULL,
    athlete_weight DECIMAL(4, 1) NOT NULL,
    athlete_place INT NOT NULL,
    CONSTRAINT pk_result PRIMARY KEY (id_result),
    CONSTRAINT chk_result_weight CHECK (athlete_weight > 0),
    CONSTRAINT chk_result_place CHECK (athlete_place >= 1),
    CONSTRAINT fk_result_athlete FOREIGN KEY (athlete_code) 
        REFERENCES athlete(athlete_code) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_result_competition FOREIGN KEY (competition_code) 
        REFERENCES competition(competition_code) ON UPDATE CASCADE ON DELETE RESTRICT
);

SELECT 
    a.surname AS athlete_surname,
    a.name AS athlete_name, 
    a.date_of_birth,
    c.appelation AS club_name,
    c.town AS club_city
FROM
    athlete a,
    trainer t,
    club c
WHERE
    a.id_trainer = t.id_trainer
    AND t.id_club = c.id_club
    AND c.town = 'Пермь'
ORDER BY
    a.surname ASC,
    a.name ASC;


USE sports_db;

SELECT
    cmp.competition_name,
    cmp.competition_date,
    a.surname AS athlete_surname,
    a.name AS athlete_name, 
    r.category,
    r.athlete_place
FROM
    result r
INNER JOIN
    athlete a ON r.athlete_code = a.athlete_code
INNER JOIN
    competition cmp ON r.competition_code = cmp.competition_code
ORDER BY
    cmp.competition_date DESC,
    r.athlete_place ASC;

USE sports_db;

SELECT
    COUNT(*) AS total_participants,
    ROUND(AVG(athlete_weight), 2) AS average_weight,
    MIN(athlete_weight) AS min_weight,
    MAX(athlete_weight) AS max_weight,
    SUM(athlete_place) AS total_places_sum
FROM
    result;


USE sports_db;

SELECT 
    t.surname AS trainer_surname,
    t.name AS trainer_name,
    t.phone AS trainer_phone,
    COUNT(a.athlete_code) AS total_athletes
FROM 
    trainer t
INNER JOIN 
    athlete a ON t.id_trainer = a.id_trainer
GROUP BY 
    t.id_trainer, 
    t.surname, 
    t.name, 
    t.phone
HAVING 
    COUNT(a.athlete_code) > 1
ORDER BY 
    total_athletes DESC;

    r.athlete_place,
    CASE 
        WHEN r.athlete_place = 1 THEN 'Золото (1 место)'
        WHEN r.athlete_place = 2 THEN 'Серебро (2 место)'
        WHEN r.athlete_place = 3 THEN 'Бронза (3 место)'

    IF(a.post IS NOT NULL, a.post, 'Почта не указана') AS email_status
FROM 
    result r
INNER JOIN 
    athlete a ON r.athlete_code = a.athlete_code
INNER JOIN 
    competition c ON r.competition_code = c.competition_code;

    SELECT AVG(athlete_weight) AS overall_avg_weight
    FROM result
)
SELECT 
    a.surname,
    a.name,
    r.athlete_weight,
    ROUND(stats.overall_avg_weight, 2) AS benchmark_avg
FROM 
    result r
INNER JOIN 
    athlete a ON r.athlete_code = a.athlete_code
CROSS JOIN 
    avg_weight_stats stats
WHERE 
    r.athlete_weight > stats.overall_avg_weight;
